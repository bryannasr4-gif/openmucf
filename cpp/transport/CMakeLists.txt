cmake_minimum_required(VERSION 3.16)
project(g4muonic_transport LANGUAGES CXX)

find_package(Geant4 REQUIRED)
option(G4MUONIC_TRANSPORT_PATCHED "Build against an overlay installation" OFF)
file(GLOB hadronic_headers "${Geant4_INCLUDE_DIRS}/G4HadronicParameters.hh")
if(NOT hadronic_headers)
  foreach(inc IN LISTS Geant4_INCLUDE_DIRS)
    if(EXISTS "${inc}/G4HadronicParameters.hh")
      list(APPEND hadronic_headers "${inc}/G4HadronicParameters.hh")
    endif()
  endforeach()
endif()
if(NOT hadronic_headers)
  message(FATAL_ERROR "G4HadronicParameters.hh was not found")
endif()
list(GET hadronic_headers 0 hadronic_header)
file(READ "${hadronic_header}" hadronic_text)
if(hadronic_text MATCHES "SetEnableMuonicData")
  set(found_patched ON)
else()
  set(found_patched OFF)
endif()
if(NOT "${found_patched}" STREQUAL "${G4MUONIC_TRANSPORT_PATCHED}")
  message(FATAL_ERROR "G4MUONIC_TRANSPORT_PATCHED disagrees with ${hadronic_header}")
endif()

add_executable(g4muonic_transport g4muonic_transport.cc)
target_compile_features(g4muonic_transport PRIVATE cxx_std_17)
target_compile_options(g4muonic_transport PRIVATE -ffp-contract=off)
target_link_libraries(g4muonic_transport PRIVATE ${Geant4_LIBRARIES})
if(G4MUONIC_TRANSPORT_PATCHED)
  target_compile_definitions(g4muonic_transport PRIVATE G4MUONIC_TRANSPORT_PATCHED=1)
endif()
