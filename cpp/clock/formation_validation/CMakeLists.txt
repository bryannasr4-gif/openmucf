cmake_minimum_required(VERSION 3.16)
project(formation_time_check LANGUAGES CXX)
find_package(Geant4 REQUIRED)
include(${Geant4_USE_FILE})
set(CLOCK_KERNEL_SOURCES "" CACHE STRING "Three enclosing clock processes")
add_library(clock_kernel OBJECT ${CLOCK_KERNEL_SOURCES})
target_compile_features(clock_kernel PRIVATE cxx_std_17)
target_link_libraries(clock_kernel PRIVATE ${Geant4_LIBRARIES})
foreach(variant IN ITEMS old new zero)
  add_library(model_${variant} OBJECT model-${variant}.cc)
  target_compile_features(model_${variant} PRIVATE cxx_std_17)
  target_link_libraries(model_${variant} PRIVATE ${Geant4_LIBRARIES})
  foreach(kind IN ITEMS wrapper transport)
    if(kind STREQUAL wrapper)
      set(probe formation_observer.cc)
    else()
      set(probe audit_transport.cc)
    endif()
    add_executable(${kind}_${variant} ${probe} $<TARGET_OBJECTS:clock_kernel> $<TARGET_OBJECTS:model_${variant}>)
    target_compile_features(${kind}_${variant} PRIVATE cxx_std_17)
    target_link_libraries(${kind}_${variant} PRIVATE ${Geant4_LIBRARIES})
    target_link_options(${kind}_${variant} PRIVATE -Wl,--export-dynamic)
  endforeach()
endforeach()
