cmake_minimum_required(VERSION 3.16)
project(stopped_muon_clock LANGUAGES CXX)
find_package(Geant4 REQUIRED)
include(${Geant4_USE_FILE})
set(CLOCK_PROCESS_SOURCES "" CACHE STRING "Optional upstream process .cc files for application object-overlay validation")
add_executable(clock_probe clock_probe.cc ${CLOCK_PROCESS_SOURCES})
target_compile_features(clock_probe PRIVATE cxx_std_17)
target_link_libraries(clock_probe PRIVATE ${Geant4_LIBRARIES})
if(UNIX)
  target_link_options(clock_probe PRIVATE -Wl,--export-dynamic)
endif()
add_executable(clock_transport clock_transport.cc ${CLOCK_PROCESS_SOURCES})
target_compile_features(clock_transport PRIVATE cxx_std_17)
target_link_libraries(clock_transport PRIVATE ${Geant4_LIBRARIES})
if(UNIX)
  target_link_options(clock_transport PRIVATE -Wl,--export-dynamic)
endif()
